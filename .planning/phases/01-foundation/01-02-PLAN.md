    ```ts
    import { create } from 'zustand'
    import { persist } from 'zustand/middleware'

    interface SidebarStore {
      collapsed: boolean
      toggle: () => void
      setCollapsed: (collapsed: boolean) => void
    }

    export const useSidebarStore = create<SidebarStore>()(
      persist(
        (set) => ({
          collapsed: false,
          toggle: () => set((state) => ({ collapsed: !state.collapsed })),
          setCollapsed: (collapsed) => set({ collapsed }),
        }),
        {
          name: 'perispek-sidebar',
          skipHydration: true, // REQUIRED: prevents SSR/client hydration mismatch in Next.js
        }
      )
    )
    ```

    **Create `src/components/layout/SidebarNav.tsx`:**
    This is a Client Component (uses usePathname hook). Nav items: Dashboard (/dashboard), Map (/map), System Insights (/insights), Work Orders (/work-orders).

    Use @phosphor-icons/react icons:
    - Dashboard → `SquaresFour` (or `ChartPieSlice`) icon
    - Map → `MapTrifold` icon
    - System Insights → `ChartLineUp` icon
    - Work Orders → `ClipboardText` icon

    Active state: `bg-surface-active text-primary` + filled icon weight. Inactive: `text-foreground-muted` + regular icon weight. Hover: `hover:bg-surface-hover hover:text-foreground`.

    When `collapsed={true}`: hide the `<span>` label (use `overflow-hidden opacity-0` with transition, or conditional render). Show only the icon. Add `title` attribute to the Link for tooltip when collapsed.

    ```tsx
    'use client'

    import Link from 'next/link'
    import { usePathname } from 'next/navigation'
    import {
      SquaresFour,
      MapTrifold,
      ChartLineUp,
      ClipboardText,
    } from '@phosphor-icons/react'
    import { cn } from '@/lib/utils/cn'

    const NAV_ITEMS = [
      { href: '/dashboard', label: 'Dashboard', icon: SquaresFour },
      { href: '/map', label: 'Map', icon: MapTrifold },
      { href: '/insights', label: 'System Insights', icon: ChartLineUp },
      { href: '/work-orders', label: 'Work Orders', icon: ClipboardText },
    ] as const

    export function SidebarNav({ collapsed }: { collapsed: boolean }) {
      const pathname = usePathname()

      return (
        <nav className="flex flex-col gap-1 px-2 py-3">
          {NAV_ITEMS.map(({ href, label, icon: Icon }) => {
            const isActive = pathname === href || pathname.startsWith(href + '/')
            return (
              <Link
                key={href}
                href={href}
                title={collapsed ? label : undefined}
                className={cn(
                  'flex items-center gap-3 rounded-md px-3 py-2.5 text-sm font-medium transition-colors duration-150',
                  isActive
                    ? 'bg-surface-active text-primary'
                    : 'text-foreground-muted hover:bg-surface-hover hover:text-foreground'
                )}
              >
                <Icon
                  size={20}
                  weight={isActive ? 'fill' : 'regular'}
                  className="shrink-0"
                />
                <span
                  className={cn(
                    'whitespace-nowrap transition-all duration-200',
                    collapsed ? 'w-0 opacity-0 overflow-hidden' : 'w-auto opacity-100'
                  )}
                >
                  {label}
                </span>
              </Link>
            )
          })}
        </nav>
      )
    }
    ```

    **Create `src/components/layout/Sidebar.tsx`:**
    Client Component. Uses useSidebarStore. Width transitions via inline style with CSS custom property (NOT Tailwind width classes — see research pitfall 6).

    Place collapse toggle button at the bottom of the sidebar. Use `CaretLeft` / `CaretRight` icon from @phosphor-icons/react to indicate direction.

    ```tsx
    'use client'

    import { useEffect } from 'react'
    import { CaretLeft, CaretRight } from '@phosphor-icons/react'
    import { useSidebarStore } from '@/lib/stores/sidebar'
    import { SidebarNav } from './SidebarNav'
    import { cn } from '@/lib/utils/cn'

    export function Sidebar() {
      const { collapsed, toggle } = useSidebarStore()

      useEffect(() => {
        // Rehydrate after mount — required to avoid SSR/client hydration mismatch
        useSidebarStore.persist.rehydrate()
      }, [])

      return (
        <aside
          className="relative flex flex-col shrink-0 bg-surface border-r border-border transition-all duration-200 ease-in-out overflow-hidden"
          style={{
            width: collapsed
              ? 'var(--sidebar-collapsed-width)'
              : 'var(--sidebar-width)',
          }}
        >
          {/* Logo / branding area */}
          <div className={cn(
            'flex items-center gap-3 px-4 border-b border-border shrink-0',
            'h-[var(--topbar-height)]'
          )}>
            {/* Logo mark — always visible */}
            <div className="w-7 h-7 rounded-md bg-primary shrink-0 flex items-center justify-center">
              <span className="text-white text-xs font-bold">P</span>
            </div>
            {/* App name — hidden when collapsed */}
            <span
              className={cn(
                'text-sm font-semibold text-foreground whitespace-nowrap transition-all duration-200',
                collapsed ? 'w-0 opacity-0 overflow-hidden' : 'w-auto opacity-100'
              )}
            >
              Perispek
            </span>
          </div>

          {/* Navigation */}
          <div className="flex-1 overflow-y-auto overflow-x-hidden">
            <SidebarNav collapsed={collapsed} />
          </div>

          {/* Collapse toggle */}
          <button
            onClick={toggle}
            className={cn(
              'flex items-center justify-center w-full h-10 border-t border-border',
              'text-foreground-muted hover:text-foreground hover:bg-surface-hover',
              'transition-colors duration-150 shrink-0'
            )}
            aria-label={collapsed ? 'Expand sidebar' : 'Collapse sidebar'}
          >
            {collapsed
              ? <CaretRight size={16} weight="bold" />
              : <CaretLeft size={16} weight="bold" />
            }
          </button>
        </aside>
      )
    }
    ```
